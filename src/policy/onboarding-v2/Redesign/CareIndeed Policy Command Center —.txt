function cx(...parts: Array<string | undefined | false | null>) {
  return parts.filter(Boolean).join(" ");
}

function pick<T>(value: T | undefined, fallback: T): T {
  return value === undefined ? fallback : value;
}

export type Intent = "neutral" | "primary" | "success" | "warning" | "danger" | "accent";
export type Size = "sm" | "md" | "lg";

function intentColor(intent: Intent) {
  switch (intent) {
    case "primary":
      return DS.color.primary;
    case "success":
      return DS.color.success;
    case "warning":
      return DS.color.warning;
    case "danger":
      return DS.color.danger;
    case "accent":
      return DS.color.accent;
    default:
      return DS.color.border;
  }
}

export function Container(props: {
  children: React.ReactNode;
  maxWidth?: number;
  paddingY?: number;
  paddingX?: number;

  const py = props.paddingY ?? DS.space[8];
  return (
    <div className={props.className} style={{ width: "100%", ...props.style }}>

export function Stack(props: {

  gap?: number;

export const Button = React.forwardRef<
  HTMLButtonElement,
  {
    children: React.ReactNode;
    intent?: Intent;
    size?: Size;
    variant?: "solid" | "outline" | "ghost";
    leftIcon?: React.ReactNode;
    rightIcon?: React.ReactNode;
    disabled?: boolean;
    onClick?: React.MouseEventHandler<HTMLButtonElement>;
    type?: "button" | "submit" | "reset";
    style?: CSS;
    className?: string;
  }
>(function Button(
  { children, intent = "primary", size = "md", variant = "solid", leftIcon, rightIcon, disabled, ...rest },
  ref
) {
  const c = intentColor(intent);

  const height = size === "sm" ? 34 : size === "lg" ? 44 : 38;
  const px = size === "sm" ? 12 : size === "lg" ? 16 : 14;
  const fontSize = size === "sm" ? 13 : 14;

  const base: CSS = {
    height,
    padding: `0 ${px}px`,
    borderRadius: DS.radius.md,
    border: "1px solid transparent",
    fontSize,
    fontWeight: 700,
    letterSpacing: "0.01em",
    cursor: disabled ? "not-allowed" : "pointer",
    opacity: disabled ? 0.6 : 1,
    display: "inline-flex",
    alignItems: "center",
    justifyContent: "center",
    gap: 10,
    userSelect: "none",
    transition: "transform 80ms ease, background 120ms ease, border-color 120ms ease, filter 120ms ease",
  };

  let style: CSS = {};
  if (variant === "solid") {
    style = {
      background: c,
      color: "#081018",
      boxShadow: "0 10px 20px rgba(0,0,0,0.25)",
    };
  } else if (variant === "outline") {
    style = {
      background: "transparent",
      color: DS.color.text,
      borderColor: c,
    };
  } else {
    style = {
      background: "transparent",
      color: DS.color.text,
      borderColor: "transparent",
    };
  }

  return (
    <button
      ref={ref}
      {...rest}
      disabled={disabled}
      className={rest.className}
      style={{ ...base, ...style, ...(rest.style ?? {}) }}
      onMouseDown={(e) => {
        if (disabled) return;
        (e.currentTarget as HTMLButtonElement).style.transform = "translateY(1px)";
      }}
      onMouseUp={(e) => {
        (e.currentTarget as HTMLButtonElement).style.transform = "translateY(0px)";
      }}
      onMouseLeave={(e) => {
        (e.currentTarget as HTMLButtonElement).style.transform = "translateY(0px)";
      }}
    >
      {leftIcon ? <span style={{ display: "inline-flex" }}>{leftIcon}</span> : null}
      <span>{children}</span>
      {rightIcon ? <span style={{ display: "inline-flex" }}>{rightIcon}</span> : null}
    </button>
  );
});

const NavItemRow: React.FC<{
  item: NavItem;
  activeId: string;
  setActiveId: (id: string) => void;
  depth?: number;
}> = ({ item, activeId, setActiveId, depth = 0 }) => {
  const [open, setOpen] = useState(
    item.children?.some(c => c.id === activeId || c.children?.some(cc => cc.id === activeId))
  );
  const isActive = activeId === item.id;
  const hasChildren = item.children && item.children.length > 0;

  return (
    <div>
      <div
        onClick={() => {
          if (hasChildren) setOpen(!open);
          else setActiveId(item.id);
        }}
        className={`
          group flex items-center gap-3 px-3 py-2.5 rounded-xl cursor-pointer
          transition-all duration-150 select-none
          ${depth > 0 ? 'ml-4 pl-3' : ''}
          ${isActive
            ? 'bg-white/20 text-white shadow-sm'
            : 'text-teal-100 hover:bg-white/10 hover:text-white'
          }
        `}
      >
        {depth === 0 && (
          <span className={`flex-shrink-0 ${isActive ? 'text-white' : 'text-teal-200 group-hover:text-white'}`}>
            {item.icon}
          </span>
        )}
        {depth > 0 && (
          <span className={`w-1 h-1 rounded-full flex-shrink-0 ${isActive ? 'bg-rust-400' : 'bg-teal-400/50'}`} />
        )}
        <span className={`flex-1 font-roboto text-sm ${depth === 0 ? 'font-medium' : 'font-normal'}`}>
          {item.label}
        </span>
        {item.badge !== undefined && (
          <span className="flex-shrink-0 min-w-[20px] h-5 px-1.5 rounded-full bg-rust-500 text-white text-[10px] font-montserrat font-bold flex items-center justify-center">
            {item.badge}
          </span>
        )}
        {hasChildren && (
          <span className="text-teal-300 flex-shrink-0">
            {open ? <ChevronDown size={13} /> : <ChevronRight size={13} />}
          </span>
        )}
      </div>

      {hasChildren && open && (
        <div className="mt-1 space-y-0.5">
          {item.children!.map(child => (
            <NavItemRow
              key={child.id}
              item={child}
              activeId={activeId}
              setActiveId={setActiveId}
              depth={depth + 1}
            />
          ))}
        </div>
      )}
    </div>
  );
};

type PolicyStatus = "Active" | "Draft" | "Needs Review" | "Archived";

type PolicyRow = {
  id: string;

  status: PolicyStatus;
  owner: string;
  lastUpdated: string;
  nextReview: string;

function statusIntent(s: PolicyStatus) {
  switch (s) {
    case "Active":
      return "success" as const;
    case "Draft":
      return "neutral" as const;
    case "Needs Review":
      return "warning" as const;
    case "Archived":
      return "danger" as const;
  }
}

function riskIntent(r: PolicyRow["risk"]) {
  switch (r) {
    case "Low":
      return "success" as const;
    case "Medium":
      return "warning" as const;
    case "High":
      return "danger" as const;
  }
}

// ============================================================
// APP ROOT — Routing and shell integration
// ============================================================

import React from 'react';
import { BrowserRouter, Routes, Route } from 'react-router-dom';
import { AppShell } from './components/AppShell';
import { Dashboard }         from './pages/Dashboard';
import { PolicyLibrary }     from './pages/PolicyLibrary';
import { PolicyDetail }      from './pages/PolicyDetail';
import { Governance }        from './pages/Governance';
import { WorkflowDrafts, WorkflowReview, WorkflowApproval } from './pages/WorkflowPages';
import { MasterCalendar }    from './pages/MasterCalendar';
import { AppendixGoverningBodyRoster } from './pages/AppendixView';

// ─── Print-only route wrapper (no shell) ──────────────────────
const PrintRoute: React.FC<{ children: React.ReactNode }> = ({ children }) => (
  <div className="print-layout">{children}</div>
);

export default function App() {
  return (
    <BrowserRouter>
      <Routes>
        {/* ── Main app routes (with shell) ── */}
        <Route path="/" element={
          <AppShell activePage="dashboard"><Dashboard /></AppShell>
        } />
        <Route path="/library" element={
          <AppShell activePage="library"><PolicyLibrary /></AppShell>
        } />
        <Route path="/library/:id" element={
          <AppShell activePage="library"><PolicyDetail /></AppShell>
        } />
        <Route path="/governance" element={
          <AppShell activePage="governance"><Governance /></AppShell>
        } />
        <Route path="/governance/body" element={
          <AppShell activePage="governance-body"><Governance /></AppShell>
        } />
        <Route path="/drafts" element={
          <AppShell activePage="drafts"><WorkflowDrafts /></AppShell>
        } />
        <Route path="/review" element={
          <AppShell activePage="review"><WorkflowReview /></AppShell>
        } />
        <Route path="/approval" element={
          <AppShell activePage="approval"><WorkflowApproval /></AppShell>
        } />
        <Route path="/calendar" element={
          <AppShell activePage="calendar"><MasterCalendar /></AppShell>
        } />

        {/* ── Appendix/form routes (with shell) ── */}
        <Route path="/library/:id/appendix/:letter" element={
          <AppShell activePage="library"><AppendixGoverningBodyRoster /></AppShell>
        } />

        {/* ── Print-only routes (no shell, print CSS fires) ── */}
        <Route path="/print/:id" element={
          <PrintRoute><PolicyDetail /></PrintRoute>
        } />
        <Route path="/print/:id/appendix/:letter" element={
          <PrintRoute><AppendixGoverningBodyRoster /></PrintRoute>
        } />
      </Routes>
    </BrowserRouter>
  );
}

Print	Browser dump, column collapse	Dedicated print architecture with @page, fixed <colgroup>, no glass
